```typescript
// Album — new fields on the existing shape (pushLibraryAlbums, pushLibraryArtistAlbums)
interface Album {
  // ...existing fields unchanged...
  badge?: string;      // disambiguation badge for a title+artist duplicate group: quality string,
                        // "Disc N", or an uppercased source ("USB"/"LOCAL"/"NAS") — whichever
                        // actually differs within the group, in that precedence order. Absent when
                        // the album is unique or nothing tracked differs within its duplicate group.
  discCount?: number;  // >1 when this Album represents a collapsed multi-disc box set (BROWSE-07);
                        // absent/0 for a normal single-disc album. When present, `uri` points at the
                        // box set's common parent directory — library:album:tracks against that uri
                        // returns ALL discs' tracks combined (MPD `search base` is recursive).
}

// Track — new field on the existing shape (pushLibraryAlbumTracks)
interface Track {
  // ...existing fields unchanged...
  disc?: number;  // MPD Disc tag, 1-based. Present on every track; omitted (0) only when MPD had no
                   // Disc tag. Tracks are pre-sorted by (disc, trackNumber, title) — clients should
                   // group consecutive same-disc runs under a "Disc N" header for a multi-disc album
                   // rather than re-sort client-side.
}

// ArtistAlbumsResponse — new field (pushLibraryArtistAlbums)
interface ArtistAlbumsResponse {
  artist: string;
  albums: Album[];
  pagination: Pagination;
  looseTracks?: Track[];  // ARTIST-04/BROWSE-04: populated ONLY when `albums` is empty — a playable
                            // song list for an artist whose tracks belong to no proper album. Absent
                            // in the normal case (every artist in the live library today has this
                            // absent — see CONTEXT.md D-08/D-09).
}
```
